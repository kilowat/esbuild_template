import { reactive } from 'uhtml/reactive';
import { effect } from '@preact/signals-core';
import { State } from './state';

type StateValue<S> = S extends State<infer T> ? T : never;
type ResolveValue<T, Context> = T | ((context: Context) => T);

type StateContext<S extends State<any>> = {
    state: S;
};

type Context<S extends State<any>, C, A> = {
    state: S;
    computed: C;
    actions: A;
    slots: Record<string, Node[]>;
    element: CustomHtmlElement<S, C, A>;
};

type ComponentInstanceContext<S extends State<any>, C, A> = Omit<Context<S, C, A>, 'element' | 'slots'> & {
    tagName: string;
};

type ComponentHooks<S extends State<any>, C, A> = {
    connected?: (context: Context<S, C, A>) => void;
    disconnected?: (context: Context<S, C, A>) => void;
    render?: (context: Context<S, C, A>) => unknown;
    listen?: (params: ListenerParams<S, C, A>) => void;
    attributeChanged?: (params: AttributeChangeCallback<S, C, A>) => void;
    computed?: ResolveValue<C, StateContext<S>>;
    actions?: ResolveValue<A, StateContext<S>>;
}

export interface ComponentOptions<
    S extends State<any>,
    C = Record<string, unknown>,
    A = Record<string, unknown>,
> extends ComponentHooks<S, C, A> {
    tagName: string;
    state: S;
    observedAttributes?: string[];
}

interface ListenerParams<S extends State<any>, C, A> extends Context<S, C, A> {
    newValue: StateValue<S>;
    oldValue: StateValue<S>;
}

interface AttributeChangeCallback<S extends State<any>, C, A> extends Context<S, C, A> {
    name: string;
    oldValue: string | null;
    newValue: string | null;
}

export interface CustomHtmlElement<
    S extends State<any> = State<any>,
    C = Record<string, unknown>,
    A = Record<string, unknown>
> extends HTMLElement {
    readonly context: Context<S, C, A>;
    readonly state: S;
    readonly computed: C;
    readonly actions: A;
    subscribeToState(callback: (params: ListenerParams<S, C, A>) => void): () => void;
    emitEvent<T = any>(name: string, detail: T): void;
}

export function createComponent<
    S extends State<any>,
    C extends Record<string, unknown> = Record<string, unknown>,
    A extends Record<string, unknown> = Record<string, unknown>,
>({
    tagName,
    connected,
    render,
    disconnected,
    attributeChanged,
    observedAttributes = [],
    listen,
    state,
    computed = {} as ResolveValue<C, StateContext<S>>,
    actions = {} as ResolveValue<A, StateContext<S>>,
}: ComponentOptions<S, C, A>): ComponentInstanceContext<S, C, A> {
    if (customElements.get(tagName)) {
        throw new Error(`Component with tag name "${tagName}" is already defined`);
    }

    const uRender = reactive(effect);

    class CustomElement extends HTMLElement implements CustomHtmlElement<S, C, A> {
        private computedValue: C;
        private actionsValue: A;
        private slotContent: Record<string, Node[]> = {};
        private renderDisposer?: ReturnType<typeof uRender>;
        private listenerDisposer?: ReturnType<typeof effect>;
        private subscribeDisposer: ReturnType<typeof effect>[] = [];

        constructor() {
            super();
            const baseContext: StateContext<S> = {
                state: this.state
            };

            this.computedValue = typeof computed === "function"
                ? computed(baseContext)
                : computed;

            this.actionsValue = typeof actions === "function"
                ? actions(baseContext)
                : actions;
        }

        public get state(): S {
            return state;
        }

        public get actions(): A {
            return this.actionsValue;
        }

        public get computed(): C {
            return this.computedValue;
        }

        static get observedAttributes() {
            return observedAttributes;
        }

        public get context(): Context<S, C, A> {
            return {
                state: this.state,
                computed: this.computedValue,
                actions: this.actionsValue,
                slots: this.slotContent,
                element: this as CustomHtmlElement<S, C, A>,
            };
        }

        public subscribeToState(callback: (params: ListenerParams<S, C, A>) => void): () => void {
            if (!state) return () => { };
            const disposer = this.setupEffect(state, callback);
            this.subscribeDisposer.push(disposer);
            return () => {
                const index = this.subscribeDisposer.indexOf(disposer);
                if (index !== -1) {
                    this.subscribeDisposer.splice(index, 1);
                }
                disposer();
            };
        }

        public emitEvent<T = any>(name: string, detail: T): void {
            this.dispatchEvent(new CustomEvent(name, { detail }));
        }

        private setupEffect(
            stateSignal: State<S>,
            callback: (params: ListenerParams<S, C, A>) => void
        ): ReturnType<typeof effect> {
            let previousValue = stateSignal.peek();
            return effect(() => {
                const currentValue = stateSignal.value;
                callback({
                    newValue: currentValue.value,
                    oldValue: previousValue.value,
                    ...this.context
                });
                previousValue = currentValue;
            });
        }

        private collectSlots() {
            const slots = {
                default: [] as Node[],
                named: {} as Record<string, Node[]>
            };

            const processElementNode = (node: Element) => {
                const slotName = node.getAttribute('data-slot');
                if (!slotName) {
                    slots.default.push(node);
                    return;
                }
                if (!slots.named[slotName]) {
                    slots.named[slotName] = [];
                }
                slots.named[slotName].push(node);
            };

            const processNode = (node: Node) => {
                if (node.nodeType === Node.ELEMENT_NODE) {
                    processElementNode(node as Element);
                } else if (node.nodeType === Node.TEXT_NODE && node.textContent?.trim()) {
                    slots.default.push(node);
                }
            };

            Array.from(this.childNodes).forEach(processNode);

            this.slotContent = {
                ...slots.named,
                default: slots.default
            };
        }

        connectedCallback() {
            try {
                requestAnimationFrame(() => {
                    this.collectSlots();
                    connected?.(this.context);
                    this.setupListener();
                    this.setupRender();
                });
            } catch (error) {
                console.error(`Error in ${tagName} connectedCallback:`, error);
            }
        }

        private setupRender() {
            if (!render) return;
            this.renderDisposer = uRender(this, () => render(this.context));
        }

        private setupListener() {
            if (!state || !listen) return;
            this.listenerDisposer = this.setupEffect(state, listen);
        }

        disconnectedCallback() {
            try {
                this.renderDisposer?.();
                this.listenerDisposer?.();
                this.subscribeDisposer.forEach(unsub => unsub());
                this.subscribeDisposer = [];
                disconnected?.(this.context);
            } catch (error) {
                console.error(`Error in ${tagName} disconnectedCallback:`, error);
            }
        }

        attributeChangedCallback(name: string, oldValue: string | null, newValue: string | null) {
            try {
                attributeChanged?.({
                    ...this.context,
                    name,
                    oldValue,
                    newValue,
                });
            } catch (error) {
                console.error(`Error in ${tagName} attributeChangedCallback:`, error);
            }
        }
    }

    customElements.define(tagName, CustomElement);

    // Initialize and return context
    const baseContext: StateContext<S> = { state };
    const computedValue = typeof computed === "function" ? computed(baseContext) : computed;
    const actionsValue = typeof actions === "function" ? actions(baseContext) : actions;

    return {
        tagName,
        state,
        computed: computedValue,
        actions: actionsValue,
    };
}
