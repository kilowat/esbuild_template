import { reactive } from 'uhtml/reactive';
import { effect } from '@preact/signals-core';
import { ComputedResult, State, compute } from './state';

type ComputedFn<S, R> = (state: S) => R;

// Modified to accept direct key-value pairs
export type ComputedProperties<S> = Record<string, ComputedResult<any>>;

export type ActionProperties<S, C> = Record<string, (...args: any[]) => any>;

type ComputedReturnTypes<S, C extends ComputedProperties<S>> = {
    [K in keyof C]: C[K];
};

type ActionReturnTypes<S, C, A extends ActionProperties<S, C>> = {
    [K in keyof A]: A[K];
};

type StateValue<S> = S extends State<infer T> ? T : S;

type StateContext<S> = {
    state: State<S>;
};

type ComputedContext<S, C> = {
    state: State<S>;
    computed: C;
};

type Context<S, C extends ComputedReturnTypes<StateValue<S>, ComputedProperties<StateValue<S>>>, A extends ActionReturnTypes<StateValue<S>, C, ActionProperties<StateValue<S>, C>>> = {
    state: State<S>;
    computed: C;
    actions: A;
    slots: Record<string, Node[]>;
    element: CustomHtmlElement<S, C, A>;
};

interface ListenerParams<S, C extends ComputedReturnTypes<StateValue<S>, ComputedProperties<StateValue<S>>>, A extends ActionReturnTypes<StateValue<S>, C, ActionProperties<StateValue<S>, C>>> extends Context<S, C, A> {
    newValue: S;
    oldValue: S;
}

interface AttributeChangeCallback<S, C extends ComputedReturnTypes<StateValue<S>, ComputedProperties<StateValue<S>>>, A extends ActionReturnTypes<StateValue<S>, C, ActionProperties<StateValue<S>, C>>> extends Context<S, C, A> {
    name: string;
    oldValue: string | null;
    newValue: string | null;
}

type ComponentHooks<S, C extends ComputedReturnTypes<StateValue<S>, ComputedProperties<StateValue<S>>>, A extends ActionReturnTypes<StateValue<S>, C, ActionProperties<StateValue<S>, C>>> = {
    connected?: (context: Context<S, C, A>) => void;
    disconnected?: (context: Context<S, C, A>) => void;
    render?: (context: Context<S, C, A>) => unknown;
    listen?: (params: ListenerParams<S, C, A>) => void;
    attributeChanged?: (params: AttributeChangeCallback<S, C, A>) => void;
}

export interface ComponentOptions<
    S,
    C extends ComputedProperties<StateValue<S>> = ComputedProperties<StateValue<S>>,
    A extends ActionProperties<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>> = ActionProperties<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>>
> extends ComponentHooks<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A>> {
    tagName: string;
    observedAttributes?: string[];
    state?: () => State<StateValue<S>>;
    computed?: (context: StateContext<StateValue<S>>) => C;
    actions?: (context: ComputedContext<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>>) => A;
}

export interface CustomHtmlElement<
    S,
    C extends ComputedReturnTypes<StateValue<S>, ComputedProperties<StateValue<S>>> = ComputedReturnTypes<StateValue<S>, ComputedProperties<StateValue<S>>>,
    A extends ActionReturnTypes<StateValue<S>, C, ActionProperties<StateValue<S>, C>> = ActionReturnTypes<StateValue<S>, C, ActionProperties<StateValue<S>, C>>
> extends HTMLElement {
    readonly context: Context<S, C, A>;
    readonly state: State<S>;
    readonly computed: C;
    readonly actions: A;
    subscribeToState(callback: (params: ListenerParams<S, C, A>) => void): () => void;
    emitEvent<T = any>(name: string, detail: T): void;
}

export function createComponent<
    S,
    C extends ComputedProperties<StateValue<S>> = ComputedProperties<StateValue<S>>,
    A extends ActionProperties<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>> = ActionProperties<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>>
>({
    tagName,
    state,
    computed: createComputed = () => ({} as C),
    actions: createActions = () => ({} as A),
    connected,
    disconnected,
    render,
    listen,
    attributeChanged,
    observedAttributes = [],
}: ComponentOptions<S, C, A>) {
    if (customElements.get(tagName)) {
        throw new Error(`Component with tag name "${tagName}" is already defined`);
    }

    const uRender = reactive(effect);
    const instances = new WeakMap<CustomElement, {
        state: State<StateValue<S>>;
        computed: ComputedReturnTypes<StateValue<S>, C>;
        actions: ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A>;
    }>();

    class CustomElement extends HTMLElement implements CustomHtmlElement<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A>> {
        private slotContent: Record<string, Node[]> = {};
        private renderDisposer?: ReturnType<typeof uRender>;
        private listenerDisposer?: ReturnType<typeof effect>;
        private subscribeDisposer: ReturnType<typeof effect>[] = [];

        constructor() {
            super();

            const createdState = (state && typeof state === "function" ? state() : undefined) as State<StateValue<S>>;

            const computedContext = {
                state: createdState
            };

            // Compute values are now directly passed in
            const computed = createComputed(computedContext) as ComputedReturnTypes<StateValue<S>, C>;

            // Pass both state and computed to actions
            const actionsContext = {
                state: createdState,
                computed
            };

            const actions = createActions(actionsContext);

            instances.set(this, {
                state: createdState,
                computed,
                actions
            });
        }

        // Rest of the implementation remains the same as before
        public get state(): State<StateValue<S>> {
            return instances.get(this)!.state;
        }

        public get computed(): ComputedReturnTypes<StateValue<S>, C> {
            return instances.get(this)!.computed;
        }

        public get actions(): ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A> {
            return instances.get(this)!.actions;
        }

        public get context(): Context<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A>> {
            return {
                state: this.state,
                computed: this.computed,
                actions: this.actions,
                slots: this.slotContent,
                element: this,
            };
        }

        static get observedAttributes() {
            return observedAttributes;
        }

        public subscribeToState(callback: (params: ListenerParams<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A>>) => void): () => void {
            const disposer = this.setupEffect(callback);
            this.subscribeDisposer.push(disposer);
            return () => {
                const index = this.subscribeDisposer.indexOf(disposer);
                if (index !== -1) {
                    this.subscribeDisposer.splice(index, 1);
                }
                disposer();
            };
        }

        public emitEvent<T = any>(name: string, detail: T): void {
            this.dispatchEvent(new CustomEvent(name, { detail }));
        }

        private setupEffect(
            callback: (params: ListenerParams<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A>>) => void
        ): ReturnType<typeof effect> {
            let previousValue = this.state.peek();
            return effect(() => {
                const currentValue = this.state.value;
                callback({
                    newValue: currentValue,
                    oldValue: previousValue,
                    ...this.context
                });
                previousValue = currentValue;
            });
        }

        private collectSlots() {
            const slots = {
                default: [] as Node[],
                named: {} as Record<string, Node[]>
            };

            const processElementNode = (node: Element) => {
                const slotName = node.getAttribute('data-slot');

                if (!slotName) {
                    slots.default.push(node);
                    return;
                }
                if (!slots.named[slotName]) {
                    slots.named[slotName] = [];
                }
                slots.named[slotName].push(node);
            };

            const processNode = (node: Node) => {
                if (node.nodeType === Node.ELEMENT_NODE) {
                    processElementNode(node as Element);
                } else if (node.nodeType === Node.TEXT_NODE && node.textContent?.trim()) {
                    slots.default.push(node);
                }
            };

            Array.from(this.childNodes).forEach(processNode);

            this.slotContent = {
                ...slots.named,
                default: slots.default
            };
        }

        connectedCallback() {
            try {
                requestAnimationFrame(() => {
                    this.collectSlots();
                    connected?.(this.context);
                    this.setupListener();
                    this.setupRender();
                });
            } catch (error) {
                console.error(`Error in ${tagName} connectedCallback:`, error);
            }
        }

        private setupRender() {
            if (!render) return;
            this.renderDisposer = uRender(this, () => render(this.context));
        }

        private setupListener() {
            if (!listen) return;
            this.listenerDisposer = this.setupEffect(listen);
        }

        disconnectedCallback() {
            try {
                this.renderDisposer?.();
                this.listenerDisposer?.();
                this.subscribeDisposer.forEach(unsub => unsub());
                this.subscribeDisposer = [];
                disconnected?.(this.context);
                instances.delete(this);
            } catch (error) {
                console.error(`Error in ${tagName} disconnectedCallback:`, error);
            }
        }

        attributeChangedCallback(name: string, oldValue: string | null, newValue: string | null) {
            try {
                attributeChanged?.({
                    ...this.context,
                    name,
                    oldValue,
                    newValue,
                } as AttributeChangeCallback<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, ActionReturnTypes<StateValue<S>, ComputedReturnTypes<StateValue<S>, C>, A>>);
            } catch (error) {
                console.error(`Error in ${tagName} attributeChangedCallback:`, error);
            }
        }
    }

    customElements.define(tagName, CustomElement);

    return { tagName };
}
